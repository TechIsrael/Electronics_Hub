 void sendData(uint8_t address, uint8_t value)
 {
    digitalWrite(SPI_SS, LOW);
    SPI.transfer(address);
    SPI.transfer(value);
    digitalWrite(SPI_SS, HIGH);
 }

void displayclear()
{
 for(int i =1; i<9;i++)

void alarmISR(){
  alarmTriggered = true;
}

void MAX72xx_begin()
{
  SPI.setBitOrder(MSBFIRST);
  sendData(DISPLAY_TEST, 0x01);
  delay(1000);
  sendData(DISPLAY_TEST, 0x00);
  sendData(DECODE_MODE, 0xFF);
  sendData(INTENSITY, 0x07);
  sendData(SCAN_LIMIT, 0x07);
  sendData(SHUTDOWN, 0x01);
}
void setup() {
  // set the slaveSelectPin as an output:
  pinMode(SPI_SS, OUTPUT);
    pinMode(DS3231_alarmPin, INPUT_PULLUP);
  pinMode(InputPin, INPUT);
  pinMode(AlarmPin, OUTPUT);
  //pinMode(DS3231_alarmPin, INPUT);
  Serial.begin(9600);
  // initialize SPI:
  SPI.begin();
  //initialise MAX72xx
  MAX72xx_begin();
  // Start the I2C interface
  Wire.begin();

  if (!rtc.begin()) {
    Serial.println("Couldn't find RTC");
    while (1); // Stop the program if RTC is not found
  }
  // Optional: Set the time to the time the sketch was compiled
  // Uncomment the line below the very FIRST time you upload the code to set the time, then comment it out and re-upload.
  //rtc.adjust(DateTime(F(__DATE__), F(__TIME__))); 
  //rtc.adjust(DateTime(2026 ,1 ,24, 15,28,10));
  // disable sware wavw and enable alarms
  rtc.disable32K();
  rtc.writeSqwPinMode(DS3231_OFF);
  // clear any existing alarms
  rtc.clearAlarm(1);
  rtc.clearAlarm(2);
  // set alarm1 (HH:MM:SS)every day 
  //rtc.setAlarm2( DateTime(0,0,0,10,30,0),DS3231_A2_Hour); //DS3231_A2_PerMinute Minute HourMinute Hour Date Day -Modes
  //rtc.enableAlarm(1);
  attachInterrupt(digitalPinToInterrupt(2),alarmISR, FALLING);
  Serial.println("DS3231 Alarm-2 Interrupt Set!");
  digitalWrite(AlarmPin, LOW);
}

   Mt = now.month(), DEC;

   //number_to_string(H,Tim);
  displayclear();

  //Time();

  if(alarmTriggered)
