 void sendData(uint8_t address, uint8_t value)
 {
    digitalWrite(SPI_SS, LOW);
    SPI.transfer(address);
    SPI.transfer(value);
    digitalWrite(SPI_SS, HIGH);
 }

{

void displayClear()
{
  sendData(9,0x00);
 for(int i =1; i<9;i++)
   {
      sendData(i,0x00);
   }
   sendData(9,0x0F);
}


void MAX72xx_begin()
{
  SPI.setBitOrder(MSBFIRST);
  sendData(DISPLAY_TEST, 0x01);
  delay(1000);
  sendData(DISPLAY_TEST, 0x00);
  sendData(DECODE_MODE, 0xFF);
  sendData(INTENSITY, 0x05);
  sendData(SCAN_LIMIT, 0x07);
  sendData(SHUTDOWN, 0x01);
}


void setup() {
  pinMode(SPI_SS, OUTPUT);
  Serial.begin(9600);
  SPI.begin();
  MAX72xx_begin();
  Wire.begin();

    while (1); // Stop the program if RTC is not found

void loop() {

  Serial.print(now.year(), DEC);

  Serial.print('/');
  Serial.print(now.day(), DEC);
  //Serial.print('/');
  Serial.println("\n");
  Serial.print(now.hour(), DEC);
  Serial.print(':');
  //if (now.minute() < 10) lcd.print('0'); // Add leading zero

}
